#include <stdio.h>
#include <stdlib.h>
int n, head, size;
int req[50];
void sort(int a[])
{
    int i, j, t;
    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - i - 1; j++)
            if (a[j] > a[j + 1])
            {
                t = a[j];
                a[j] = a[j + 1];
                a[j + 1] = t;
            }
}
void sstf()
{
    int visited[50] = {0};
    int i, j, pos = head, min, index, total = 0;
    printf("\nSSTF order: %d", head);
    for (i = 0; i < n; i++)
    {
        min = 9999;
        for (j = 0; j < n; j++)
            if (visited[j] == 0 && abs(req[j] - pos) < min)
            {
                min = abs(req[j] - pos);
                index = j;
            }
        visited[index] = 1;
        total = total + min;
        pos = req[index];
        printf(" -> %d", pos);
    }
    printf("\nTotal head movement = %d\n", total);
}
void scan()
{
    int a[50], i, j, pos = head, total = 0;
    for (i = 0; i < n; i++)
        a[i] = req[i];
    sort(a);
    for (i = 0; i < n; i++)
        if (a[i] >= head)
            break;
    printf("\nSCAN order: %d", head);
    for (j = i; j < n; j++)
    {
        printf(" -> %d", a[j]);
        total = total + abs(a[j] - pos);
        pos = a[j];
    }
    printf(" -> %d", size - 1);
    total = total + (size - 1 - pos);
    pos = size - 1;
    for (j = i - 1; j >= 0; j--)
    {
        printf(" -> %d", a[j]);
        total = total + abs(pos - a[j]);
        pos = a[j];
    }
    printf("\nTotal head movement = %d\n", total);
}
void clook()
{
    int a[50], i, j, pos = head, total = 0;
    for (i = 0; i < n; i++)
        a[i] = req[i];
    sort(a);
    for (i = 0; i < n; i++)
        if (a[i] >= head)
            break;
    printf("\nC-LOOK order: %d", head);
    for (j = i; j < n; j++)
    {
        printf(" -> %d", a[j]);
        total = total + abs(a[j] - pos);
        pos = a[j];
    }
    for (j = 0; j < i; j++)
    {
        printf(" -> %d", a[j]);
        total = total + abs(pos - a[j]);
        pos = a[j];
    }
    printf("\nTotal head movement = %d\n", total);
}
int main()
{
    int i, ch;
    printf("Enter number of requests: ");
    scanf("%d", &n);
    printf("Enter request queue: ");
    for (i = 0; i < n; i++)
        scanf("%d", &req[i]);
    printf("Enter initial head position: ");
    scanf("%d", &head);
    printf("Enter total number of tracks: ");
    scanf("%d", &size);
    do
    {
        printf("\n1. SSTF\n2. SCAN\n3. C-LOOK\n4. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &ch);
        if (ch == 1)
            sstf();
        else if (ch == 2)
            scan();
        else if (ch == 3)
            clook();
    } while (ch != 4);
    return 0;
}
