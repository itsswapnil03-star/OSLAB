#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
void bubble(int a[], int n)
{
    int i, j, t;
    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - i - 1; j++)
            if (a[j] > a[j + 1])
            {
                t = a[j];
                a[j] = a[j + 1];
                a[j + 1] = t;
            }
}
void selection(int a[], int n)
{
    int i, j, min, t;
    for (i = 0; i < n - 1; i++)
    {
        min = i;
        for (j = i + 1; j < n; j++)
            if (a[j] < a[min])
                min = j;
        t = a[i];
        a[i] = a[min];
        a[min] = t;
    }
}
void sort_demo()
{
    int a[50], n, i, pid;
    printf("Enter number of integers: ");
    scanf("%d", &n);
    printf("Enter %d integers: ", n);
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);
    pid = fork();
    if (pid == 0)
    {
        printf("\nChild process (PID %d)\n", getpid());
        selection(a, n);
        printf("Sorted by child: ");
        for (i = 0; i < n; i++)
            printf("%d ", a[i]);
        printf("\n");
    }
    else
    {
        printf("\nParent process (PID %d)\n", getpid());
        bubble(a, n);
        printf("Sorted by parent: ");
        for (i = 0; i < n; i++)
            printf("%d ", a[i]);
        printf("\n");
        wait(NULL);
        printf("Child finished\n");
    }
}
void zombie_demo()
{
    int pid;
    pid = fork();
    if (pid == 0)
    {
        printf("Child PID %d exiting\n", getpid());
        exit(0);
    }
    else
    {
        printf("Parent PID %d sleeping for 30 seconds\n", getpid());
        printf("Run ps -l in another terminal to see the zombie (Z)\n");
        sleep(30);
        printf("Parent exiting\n");
    }
}
void orphan_demo()
{
    int pid;
    pid = fork();
    if (pid == 0)
    {
        printf("Child PID %d, Parent PID %d\n", getpid(), getppid());
        sleep(10);
        printf("Child PID %d, New Parent PID %d\n", getpid(), getppid());
    }
    else
    {
        printf("Parent PID %d exiting\n", getpid());
        exit(0);
    }
}
int main()
{
    int ch;
    printf("1. Sort using fork and wait\n");
    printf("2. Zombie state\n");
    printf("3. Orphan state\n");
    printf("Enter your choice: ");
    scanf("%d", &ch);
    if (ch == 1)
        sort_demo();
    else if (ch == 2)
        zombie_demo();
    else if (ch == 3)
        orphan_demo();
    else
        printf("Invalid choice\n");
    return 0;
}
