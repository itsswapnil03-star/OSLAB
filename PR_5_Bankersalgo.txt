#include <stdio.h>
int main()
{
    int n, m, i, j, k;
    int alloc[10][10], max[10][10], need[10][10];
    int avail[10], work[10], finish[10], safe[10];
    int count = 0, found;
    printf("Enter number of processes: ");
    scanf("%d", &n);
    printf("Enter number of resources: ");
    scanf("%d", &m);
    printf("Enter Allocation matrix:\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < m; j++)
            scanf("%d", &alloc[i][j]);
    printf("Enter Max matrix:\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < m; j++)
            scanf("%d", &max[i][j]);
    printf("Enter Available resources:\n");
    for (j = 0; j < m; j++)
        scanf("%d", &avail[j]);
    for (i = 0; i < n; i++)
        for (j = 0; j < m; j++)
            need[i][j] = max[i][j] - alloc[i][j];
    printf("\nNeed matrix:\n");
    for (i = 0; i < n; i++)
    {
        for (j = 0; j < m; j++)
            printf("%d ", need[i][j]);
        printf("\n");
    }
    for (j = 0; j < m; j++)
        work[j] = avail[j];
    for (i = 0; i < n; i++)
        finish[i] = 0;
    while (count < n)
    {
        found = 0;
        for (i = 0; i < n; i++)
        {
            if (finish[i] == 0)
            {
                for (j = 0; j < m; j++)
                    if (need[i][j] > work[j])
                        break;
                if (j == m)
                {
                    for (k = 0; k < m; k++)
                        work[k] = work[k] + alloc[i][k];
                    safe[count] = i;
                    count++;
                    finish[i] = 1;
                    found = 1;
                }
            }
        }
        if (found == 0)
            break;
    }
    if (count == n)
    {
        printf("\nSystem is in SAFE state\nSafe sequence: ");
        for (i = 0; i < n; i++)
            printf("P%d ", safe[i]);
        printf("\n");
    }
    else
        printf("\nSystem is in UNSAFE state (deadlock possible)\n");
    return 0;
}
